# ── Sieve (server-side mailregler) ───────────────────────────────────────
# sievelib: tre delar i ett paket —
#   sievelib.factory      bygger Sieve-kod programmatiskt ur regler
#   sievelib.parser       validerar genererad kod innan den skickas
#   sievelib.managesieve  klient mot Dovecot ManageSieve (RFC 5804, port 4190)
#
# KRITISK: en trasig aktiv Sieve-regel far Dovecot att avvisa eller
# felsortera post. Darfor importeras sievelib.parser ALLTID fore
# putscript — se sieve.script._validate().
#
# Versionstak: 1.x har ett stabilt API (FiltersSet.addfilter,
# Parser.parse, Client.connect). 2.x kan byta signatur — ta inte upp
# utan att testa round-trip mot en riktig server forst.
sievelib>=1.6,<2

# ── knowledge_joplin (transitivt beroende via user_mail_ai) ──────────────

# Utan den vägrar Odoo installera user_mail_ai med "External dependency
# diff-match-patch not installed" — och felet pekar på knowledge_joplin,
# inte på user_mail_ai, så det är lätt att leta på fel ställe.
diff-match-patch
